    <script>
        let the = {
            element: null,
            stepper: null,
            currentStepIndex: 0,
            modalElement: null
        };
        document.addEventListener('livewire:load', function() {
    let stepperInstances = {};
    let currentStepIndices = {};

    Livewire.hook('message.processed', (message, component) => {
        if (component.name === 'add-audit-modal-component') {
            const modalElement = document.getElementById('addAuditModalComponent');
            if (modalElement && modalElement.classList.contains('show')) {
                const stepperElement = modalElement.querySelector("#kt_modal_add_audit_stepper");
                if (stepperElement) {
                  // Ensure stepper is initialized or re-initialized
                    setTimeout(() => {
                        initializeStepper(modalElement, stepperElement);
                    }, 0);
                } else {
                    console.error("Stepper element not found after Livewire update.");
                }
            }
        }
    });

    function initializeStepper(modalElement, stepperElement) {
        const modalId = modalElement.id;

        if (!stepperInstances[modalId]) {
            stepperInstances[modalId] = new KTStepper(stepperElement);
            currentStepIndices[modalId] = 0;

            stepperInstances[modalId].on("kt.stepper.next", function(stepperObj) {
                const currentStep = currentStepIndices[modalId] + 1;
                Livewire.emit('validateStep', currentStep, (isValid) => {
                    if (!isValid) {
                        stepperObj.goPrevious(); // Prevent next step
                        return; // Very Important: Stop further execution in this callback
                    }
                    currentStepIndices[modalId]++;
                    // IMPORTANT: Use setTimeout to ensure the DOM has updated before goTo
                    setTimeout(() => {
                        stepperObj.goTo(currentStep + 1);
                    }, 0);
                    Livewire.emit('nextStep');
                });
            });

            stepperInstances[modalId].on("kt.stepper.previous", function() {
                currentStepIndices[modalId]--;
                stepperInstances[modalId].goTo(currentStepIndices[modalId] + 1);
                Livewire.emit('previousStep');
            });
        } else {
          // If already initialized, just go to the correct step
            stepperInstances[modalId].goTo(currentStepIndices[modalId] + 1);
        }
    }

    document.addEventListener('shown.bs.modal', function(event) {
        const modalElement = document.getElementById('addAuditModalComponent');
        if (modalElement) {
            const stepperElement = modalElement.querySelector("#kt_modal_add_audit_stepper");
            if (stepperElement) {
                initializeStepper(modalElement, stepperElement);
            } else {
                console.error("Stepper element not found on modal shown.");
            }
        } else {
            console.error("Modal element not found on modal shown.");
        }
    });

    document.addEventListener('hidden.bs.modal', function(event) {
        const modalElement = document.getElementById('addAuditModalComponent');
        if(modalElement){
            const modalId = modalElement.id;
            if (stepperInstances[modalId]) {
                stepperInstances[modalId].destroy();
                delete stepperInstances[modalId];
                delete currentStepIndices[modalId];
            }
        }
    });
});

        document.addEventListener('livewire:load', function() {
            let flatpickrInstances = {}; // Store instances by modal ID

            Livewire.hook('message.processed', (message, component) => {
                if (component.name === 'audit.add-audit-modal') {
                    const modalElement = document.getElementById('addAuditModalComponent');
                    if (modalElement && modalElement.classList.contains('show')) {
                        initializeFlatpickr(modalElement);
                    }
                }
            });

            function initializeFlatpickr(modalElement) {
                const modalId = modalElement.id; // Use modal ID for instance storage
                const dateInput = modalElement.querySelector('#a_date_range');

                if (!dateInput) {
                    console.error('Date input #a_date_range not found in modal', modalId);
                    return; // Important: Exit if input not found
                }

                if (!flatpickrInstances[modalId]) {
                    flatpickrInstances[modalId] = flatpickr(dateInput, {
                        mode: "range",
                        dateFormat: "Y-m-d",
                        altInput: true,
                        altFormat: "F j, Y",
                    });
                    modalElement.dataset.flatpickrInitialized = true;
                } else {
                    flatpickrInstances[modalId].destroy();
                    flatpickrInstances[modalId] = flatpickr(dateInput, {
                        mode: "range",
                        dateFormat: "Y-m-d",
                        altInput: true,
                        altFormat: "F j, Y",
                    });
                }
            }

            const modalElement = document.getElementById('addAuditModalComponent');
            if (modalElement) {
                modalElement.addEventListener('shown.bs.modal', function() {
                    initializeFlatpickr(modalElement);
                });
                modalElement.addEventListener('hidden.bs.modal', function() {
                    const modalId = modalElement.id;
                    if (flatpickrInstances[modalId]) {
                        flatpickrInstances[modalId].destroy();
                        delete flatpickrInstances[modalId];
                        modalElement.dataset.flatpickrInitialized = false;
                    }
                });
            }
        });
